================================================================================
QUICK REFERENCE - WHAT TO RUN
================================================================================

FIRST TIME:
  python run.py --setup              # Check setup, create data

TRAINING:
  python run.py --quick-train        # 10 epochs (~5-10 min on GPU)
  python run.py --full-train         # 200 epochs (~2-4 hours on GPU)
  python train.py                    # Custom (edit config.py first)

INFERENCE:
  python inference.py --create-sample
  python inference.py --input sketch.jpg --output painting.png
  python inference.py --input folder/ --output folder/ --batch

EXPERIMENTS:
  python experiments.py              # Interactive menu
  python experiments.py              # Research questions

HELP:
  python run.py --help
  python inference.py --help
